export const icon = "ZoomIn"

# {title}

`transformScale` 是 SVG 缩放动画的生成工具，可以创建围绕任意支点（pivot）缩放的 `<animateTransform type="scale">` 元素组合。

---

## 函数签名

```tsx
import { transformScale } from '@guohub8080/expub-tool/smil';

transformScale({
  initValue?: number,                        // 默认 1：初始缩放比例（1 = 原始大小）
  timeline: ScaleKeyframe[],                 // 必填：时间线段数组
  pivot: [number, number],                   // 必填：缩放支点 [cx, cy]
  begin?: string,                            // 默认 undefined：开始时机（SMIL begin 字符串）
  calcMode?: 'spline' | 'linear' | 'discrete' | 'paced',  // 自动：动画计算模式
  isFreeze?: boolean,                        // 默认 false：是否保持最终状态
  loopCount?: number,                        // 默认 1：循环次数（0 = 无限）
  isAdditive?: boolean,                      // 默认 true：是否累加变换
  restart?: 'always' | 'whenNotActive' | 'never',
  native?: T_NativeAnimateTransform,        // 透传/覆盖原生 animateTransform 属性（最后展开，同名覆盖生成值）
})
```

### ScaleKeyframe 类型

```tsx
interface ScaleKeyframe {
  toAbs: number,                             // 必填：目标缩放比例（绝对值）
  durationSeconds: number,                   // 必填：该段持续时间（秒）
  keySplines?: string,                       // 默认线性 "0 0 1 1"：贝塞尔缓动参数
}
```

### pivot 必填

新 API 用显式坐标 `pivot: [cx, cy]` 指定缩放中心（**必填**），不再支持 `'center'` 等快捷字符串，也不需要 CSS `transform-origin` 配合。
元素中心按 `x + w/2`、`y + h/2` 计算，例如 circle `cx=100 cy=100` 的 pivot 是 `[100, 100]`。

---

## 缩放比例说明

| 比例 | 效果 | 说明 |
|------|------|------|
| `0` | 完全消失 | 元素缩小到不可见 |
| `0.5` | 缩小一半 | 元素变为原始大小的 50% |
| `1` | 原始大小 | 默认状态 |
| `1.5` | 放大 1.5 倍 | 元素变为原始大小的 150% |
| `2` | 放大两倍 | 元素变为原始大小的 200% |

## 使用示例

### 简单的放大缩小

```tsx
<svg width="200" height="200">
  <circle cx="100" cy="100" r="30" fill="blue">
    {transformScale({
      initValue: 1,
      pivot: [100, 100],  // circle 中心
      timeline: [
        { toAbs: 1.5, durationSeconds: 1 },  // 放大到 1.5 倍
        { toAbs: 1, durationSeconds: 1 }     // 缩小回 1 倍
      ]
    })}
  </circle>
</svg>
```

**生成的 SVG 代码：**

```xml
<!-- 第一步：平移到支点 -->
<animateTransform
  attributeName="transform"
  type="translate"
  values="100 100; 100 100; 100 100"
  keyTimes="0; 0.5; 1"
  dur="2s"
  calcMode="linear"
  repeatCount="1"
  fill="remove"
  additive="sum"
/>
<!-- 第二步：执行缩放 -->
<animateTransform
  attributeName="transform"
  type="scale"
  values="1 1; 1.5 1.5; 1 1"
  keyTimes="0; 0.5; 1"
  dur="2s"
  calcMode="linear"
  repeatCount="1"
  fill="remove"
  additive="sum"
/>
<!-- 第三步：平移回原位置 -->
<animateTransform
  attributeName="transform"
  type="translate"
  values="-100 -100; -100 -100; -100 -100"
  keyTimes="0; 0.5; 1"
  dur="2s"
  calcMode="linear"
  repeatCount="1"
  fill="remove"
  additive="sum"
/>
```

**效果**：圆形从原始大小放大到 1.5 倍，然后缩小回原始大小

**说明**：为了围绕任意支点缩放，使用了三段变换：translate(cx,cy) → scale(sx,sy) → translate(-cx,-cy)


```xml
  fill="remove"
```

## pivot 坐标计算

缩放支点直接写 `[cx, cy]` 坐标，按元素位置自行计算。以元素 `x=70, y=70, w=60, h=60` 为例：

| 位置 | 计算方式 | pivot 值 |
|--------|----------|--------|
| 中心 | `[x + w/2, y + h/2]` | `[100, 100]` |
| 顶部中心 | `[x + w/2, y]` | `[100, 70]` |
| 右下角 | `[x + w, y + h]` | `[130, 130]` |
| 左上角 | `[x, y]` | `[70, 70]` |

## 注意事项

### isAdditive 属性

- **推荐使用 `isAdditive: true`**（默认）：动画会与其他变换叠加
- 使用 `isAdditive: false`：动画会替换其他变换

### 缩放比例限制

- 过大的缩放比例（如 10 以上）可能导致性能问题
- 负数缩放比例会翻转元素（通常不是预期效果）

### 与其他动画组合

```tsx
<circle cx="100" cy="100" r="30" fill="blue">
  {/* 同时缩放和旋转 */}
  {transformScale({
    initValue: 1,
    pivot: [100, 100],
    timeline: [{ toAbs: 1.5, durationSeconds: 2 }]
  })}
  {transformRotate({
    initValue: 0,
    pivot: [100, 100],
    timeline: [{ toAbs: 360, durationSeconds: 2 }]
  })}
</circle>
```

### isFreeze 属性

- **`isFreeze: false`**（默认）：动画结束后回到原始状态
- **`isFreeze: true`**：动画结束后停留在最后一帧

## 完整示例：交互式按钮（悬停放大）

```tsx
import { transformScale } from '@guohub8080/expub-tool/smil';

const InteractiveButton = () => {
  return (
    <svg width="200" height="60">
      <g>
        {/* 按钮背景 */}
        <rect
          x="10"
          y="10"
          width="180"
          height="40"
          rx="8"
          fill="#3b82f6"
        >
          {/* 放大动画（按钮中心 = (10 + 90, 10 + 20)） */}
          {transformScale({
            pivot: [100, 30],
            initValue: 1,
            timeline: [
              { toAbs: 1.05, durationSeconds: 0.2 }
            ],
            isFreeze: true
          })}
        </rect>

        {/* 按钮文字 */}
        <text
          x="100"
          y="36"
          textAnchor="middle"
          fill="white"
          fontSize="16"
          fontWeight="bold"
        >
          点击我
        </text>
      </g>
    </svg>
  );
};
```
